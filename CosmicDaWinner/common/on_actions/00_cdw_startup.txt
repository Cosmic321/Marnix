# Cosmic DaWinner - world setup that cannot be expressed in history files.
#
# 1. Every country is given a core on every state it owns. The state files already
#    carry add_core_of, but doing it here as well guarantees the whole world is cored
#    however the history is loaded or re-owned.
# 2. Every country starts with a modest amount of equipment, scaled to its industry.
#    Enough to fill out the opening weeks, nowhere near enough to fight a war with.

# Re-assert cores every month. Cheap, and it repairs any state a script, a peace
# conference or a console command has quietly stripped of its owner's core.
on_monthly = {
	effect = {
		every_country = {
			every_owned_state = {
				add_core_of = PREV
			}
		}
	}
}
