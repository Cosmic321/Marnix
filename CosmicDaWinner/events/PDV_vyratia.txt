add_namespace = pdv_vyratia

# The night the fascists came for the leadership of the Directorate.
country_event = {
	id = pdv_vyratia.1
	title = pdv_vyratia.1.t
	desc = pdv_vyratia.1.d
	is_triggered_only = yes

	option = {
		name = pdv_vyratia.1.a
		add_war_support = 0.05
		add_stability = -0.05
		# Only the communist twin that broke away is left without a head.
		every_country = {
			limit = {
				tag = ROOT
				has_war_with = ROOT
			}
			add_ideas = PDV_leaderless
		}
		# The provinces that walked out a week ago now declare for the communists.
		if = { limit = { SUN = { exists = yes } } SUN = { declare_war_on = { target = ROOT type = annex_everything } } }
		if = { limit = { BRR = { exists = yes } } BRR = { declare_war_on = { target = ROOT type = annex_everything } } }
		if = { limit = { AFR = { exists = yes } } AFR = { declare_war_on = { target = ROOT type = annex_everything } } }
		if = { limit = { VYR = { exists = yes } } VYR = { declare_war_on = { target = ROOT type = annex_everything } } }
		if = { limit = { FGT = { exists = yes } } FGT = { declare_war_on = { target = ROOT type = annex_everything } } }
		if = { limit = { PPS = { exists = yes } } PPS = { declare_war_on = { target = ROOT type = annex_everything } } }
	}
}

# Three years after the economy was forced to move, the bill arrives.
country_event = {
	id = pdv_vyratia.10
	title = pdv_vyratia.10.t
	desc = pdv_vyratia.10.d
	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = pdv_vyratia.10.a
		remove_ideas = PDV_forced_economy
		clr_country_flag = PDV_forced_time_started
		clr_country_flag = PDV_forced_clock_on
		if = {
			limit = {
				VRE = { exists = yes }
				NOT = { VRE = { OR = { has_capitulated = yes is_puppet_of = ROOT } } }
			}
			add_ideas = PDV_economy_falling
			add_stability = -0.10
			every_country = {
				limit = { NOT = { tag = ROOT } }
				country_event = { id = pdv_vyratia.12 days = 1 }
			}
		}
		else = {
			add_stability = 0.05
			add_political_power = 50
		}
	}
}

# and the three-year clock is put on the board.

	id = pdv_vyratia.13
	title = pdv_vyratia.13.t
	desc = pdv_vyratia.13.d
	is_triggered_only = yes
	hidden = yes

	option = {
		name = pdv_vyratia.13.a

		if = { limit = { has_idea = overstretched_industry } remove_ideas = overstretched_industry }
		if = { limit = { has_idea = overstretched_industry_triage } remove_ideas = overstretched_industry_triage }
		if = { limit = { has_idea = overstretched_industry_reformed } remove_ideas = overstretched_industry_reformed }

		add_ideas = PDV_forced_economy

		every_country = {
			country_event = { id = pdv_vyratia.11 }
		}

		activate_decision = PDV_forced_economy_mission
		country_event = { id = pdv_vyratia.10 days = 1093 }
	}
}
