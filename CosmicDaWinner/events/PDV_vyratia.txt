add_namespace = pdv_vyratia

# The night the fascists came for the leadership of the Directorate.
country_event = {
	id = pdv_vyratia.1
	title = pdv_vyratia.1.t
	desc = pdv_vyratia.1.d
	is_triggered_only = yes

	option = {
		name = pdv_vyratia.1.a
		add_war_support = 0.05
		add_stability = -0.05
		# The Directorate's leadership is gone: whoever is left is fighting without a head.
		every_country = {
			limit = {
				NOT = { tag = ROOT }
				at_war_with = ROOT
			}
			add_ideas = PDV_leaderless
		}
	}
}
